cmake_minimum_required(VERSION 3.16)
project(splinesketch LANGUAGES CXX)

add_library(splinesketch INTERFACE)
target_include_directories(splinesketch INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(splinesketch INTERFACE cxx_std_17)

option(SPLINESKETCH_BUILD_TESTS "Build the test executable" ON)
if(SPLINESKETCH_BUILD_TESTS)
  enable_testing()
  add_executable(splinesketch_certified_rank_tests tests/certified_rank_tests.cpp)
  target_link_libraries(splinesketch_certified_rank_tests PRIVATE splinesketch)
  add_test(NAME splinesketch_certified_rank_tests COMMAND splinesketch_certified_rank_tests)
  add_executable(splinesketch_certified_exception_tests tests/exception_safety_tests.cpp)
  target_link_libraries(splinesketch_certified_exception_tests PRIVATE splinesketch)
  target_compile_definitions(splinesketch_certified_exception_tests PRIVATE SPLINESKETCH_CERTIFIED)
  add_test(NAME splinesketch_certified_exception_tests COMMAND splinesketch_certified_exception_tests)
  add_executable(splinesketch_tests tests/splinesketch_tests.cpp)
  target_link_libraries(splinesketch_tests PRIVATE splinesketch)
  add_test(NAME splinesketch_tests COMMAND splinesketch_tests)
  add_executable(splinesketch_exception_safety_tests tests/exception_safety_tests.cpp)
  target_link_libraries(splinesketch_exception_safety_tests PRIVATE splinesketch)
  add_test(NAME splinesketch_exception_safety_tests COMMAND splinesketch_exception_safety_tests)
  add_executable(splinesketch_guarantee_invariant_tests tests/guarantee_invariant_tests.cpp)
  target_link_libraries(splinesketch_guarantee_invariant_tests PRIVATE splinesketch)
  target_compile_definitions(splinesketch_guarantee_invariant_tests PRIVATE SPLINESKETCH_TESTING)
  add_test(NAME splinesketch_guarantee_invariant_tests COMMAND splinesketch_guarantee_invariant_tests)
  add_executable(splinesketch_implementation_proof_tests tests/implementation_proof_tests.cpp)
  target_link_libraries(splinesketch_implementation_proof_tests PRIVATE splinesketch)
  target_compile_definitions(splinesketch_implementation_proof_tests PRIVATE SPLINESKETCH_TESTING)
  add_test(NAME splinesketch_implementation_proof_tests COMMAND splinesketch_implementation_proof_tests)
  add_executable(splinesketch_guarantee_obligation_tests tests/guarantee_obligation_tests.cpp)
  target_link_libraries(splinesketch_guarantee_obligation_tests PRIVATE splinesketch)
  target_compile_definitions(splinesketch_guarantee_obligation_tests PRIVATE SPLINESKETCH_VERIFY_GUARANTEES)
  add_test(NAME splinesketch_guarantee_obligation_tests COMMAND splinesketch_guarantee_obligation_tests)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    add_executable(splinesketch_certified_rank_binary64_tests tests/certified_rank_tests.cpp)
    target_link_libraries(splinesketch_certified_rank_binary64_tests PRIVATE splinesketch)
    target_compile_options(splinesketch_certified_rank_binary64_tests PRIVATE -mlong-double-64)
    add_test(NAME splinesketch_certified_rank_binary64_tests COMMAND splinesketch_certified_rank_binary64_tests)
    add_executable(splinesketch_implementation_proof_binary64_tests tests/implementation_proof_tests.cpp)
    target_link_libraries(splinesketch_implementation_proof_binary64_tests PRIVATE splinesketch)
    target_compile_options(splinesketch_implementation_proof_binary64_tests PRIVATE -mlong-double-64)
    target_compile_definitions(splinesketch_implementation_proof_binary64_tests PRIVATE SPLINESKETCH_TESTING)
    add_test(NAME splinesketch_implementation_proof_binary64_tests COMMAND splinesketch_implementation_proof_binary64_tests)
    add_executable(splinesketch_guarantee_numeric_obligation_tests tests/guarantee_numeric_obligation_tests.cpp)
    target_link_libraries(splinesketch_guarantee_numeric_obligation_tests PRIVATE splinesketch)
    target_compile_options(splinesketch_guarantee_numeric_obligation_tests PRIVATE -mlong-double-64)
    target_compile_definitions(splinesketch_guarantee_numeric_obligation_tests PRIVATE SPLINESKETCH_VERIFY_GUARANTEES)
    add_test(NAME splinesketch_guarantee_numeric_obligation_tests COMMAND splinesketch_guarantee_numeric_obligation_tests)
  endif()
endif()

option(SPLINESKETCH_BUILD_BENCHMARK "Build the timing benchmarks" OFF)
if(SPLINESKETCH_BUILD_BENCHMARK)
  add_executable(splinesketch_benchmark benchmarks/throughput.cpp)
  target_link_libraries(splinesketch_benchmark PRIVATE splinesketch)
  add_executable(splinesketch_criteria_benchmark benchmarks/criteria.cpp)
  target_link_libraries(splinesketch_criteria_benchmark PRIVATE splinesketch)
endif()

if(SPLINESKETCH_BUILD_TESTS OR SPLINESKETCH_BUILD_BENCHMARK)
  add_executable(splinesketch_accuracy_benchmark benchmarks/accuracy.cpp)
  target_link_libraries(splinesketch_accuracy_benchmark PRIVATE splinesketch)
  add_executable(splinesketch_certified_accuracy_benchmark benchmarks/accuracy.cpp)
  target_link_libraries(splinesketch_certified_accuracy_benchmark PRIVATE splinesketch)
  target_compile_definitions(splinesketch_certified_accuracy_benchmark PRIVATE SPLINESKETCH_CERTIFIED)
  if(SPLINESKETCH_BUILD_TESTS)
    add_test(NAME splinesketch_certified_accuracy_benchmark COMMAND splinesketch_certified_accuracy_benchmark)
    add_test(NAME splinesketch_accuracy_benchmark COMMAND splinesketch_accuracy_benchmark)
  endif()
endif()
